### 虚拟摇杆详细设计

```
虚拟摇杆结构：

      ┌─────────────────┐
      │                 │
      │    ╭─────╮      │  ← 外圈（固定）
      │    │  ●  │      │     直径: 120px
      │    │     │      │     半透明背景
      │    ╰─────╯      │
      │                 │
      └─────────────────┘
            ↑
        摇杆头（可移动）
        直径: 60px
        拖动范围: ±50px

交互流程：
1. 手指按下摇杆区域
2. 摇杆头跟随手指移动
3. 坦克朝摇杆偏移方向移动
4. 手指抬起，摇杆头回中
5. 坦克停止移动

代码示例：
```javascript
class VirtualJoystick {
  constructor(container) {
    this.centerX = 60;
    this.centerY = 60;
    this.maxRadius = 50;
    this.currentAngle = 0;
    this.currentDistance = 0;
    
    this.createJoystick(container);
  }
  
  onMove(angle, distance) {
    // angle: 0-360度
    // distance: 0-1 (归一化距离)
    const speed = distance * this.maxSpeed;
    const vx = Math.cos(angle) * speed;
    const vy = Math.sin(angle) * speed;
    
    this.game.moveTank(vx, vy);
    this.game.rotateTank(angle);
  }
}
```

// 使用离屏Canvas预渲染静态元素
const offscreenCanvas = document.createElement('canvas');
const offscreenCtx = offscreenCanvas.getContext('2d');
// 绘制背景网格到离屏Canvas
renderBackground(offscreenCtx);

// 主循环只需拷贝离屏Canvas
function render() {
  ctx.drawImage(offscreenCanvas, 0, 0);
  // 绘制动态元素...
}

// 使用requestAnimationFrame限制帧率
let lastFrameTime = 0;
const targetFPS = 60;
const frameInterval = 1000 / targetFPS;

function gameLoop(timestamp) {
  if (timestamp - lastFrameTime < frameInterval) {
    requestAnimationFrame(gameLoop);
    return;
  }
  lastFrameTime = timestamp;
  update();
  render();
  requestAnimationFrame(gameLoop);
}
```
